-- -------------------------------------------------
-- DROP TABLES
-- -------------------------------------------------

DROP TABLE IF EXISTS Komentaras;
DROP TABLE IF EXISTS Progresas;
DROP TABLE IF EXISTS Ataskaita;
DROP TABLE IF EXISTS Uzduotis;
DROP TABLE IF EXISTS Resursas;
DROP TABLE IF EXISTS Renginys;
DROP TABLE IF EXISTS Projektas;
DROP TABLE IF EXISTS Klientas;
DROP TABLE IF EXISTS Narys;
DROP TABLE IF EXISTS Komanda;
DROP TABLE IF EXISTS Organizacija;
DROP TABLE IF EXISTS Vartotojas;

-- -------------------------------------------------
-- CREATE TABLES
-- -------------------------------------------------

-- Sukurti Vartotojas lentele
CREATE TABLE Vartotojas (
    VartotojoID INT PRIMARY KEY,
    Vartotojo_vardas VARCHAR(50),
    Vardas VARCHAR(50),
    Pavarde VARCHAR(50),
    Gimimo_data DATE,
    Telefono_numeris VARCHAR(15),
    El_pastas VARCHAR(100),
    Slaptazodis VARCHAR(100)
);

-- Sukurti Organizacija lentele
CREATE TABLE Organizacija (
    OrganizacijosID INT PRIMARY KEY,
    Pavadinimas VARCHAR(50),
    Skyrius VARCHAR(50)
);

-- Sukurti Komanda lentele
CREATE TABLE Komanda (
    KomandosID INT PRIMARY KEY,
	OrganizacijosID INT,
    Pavadinimas VARCHAR(50),
    Sukurimo_data DATE,
    Tipas VARCHAR(50),
    Paskirtis VARCHAR(255),
    
    FOREIGN KEY (OrganizacijosID) REFERENCES Organizacija (OrganizacijosID)
);

-- Sukurti Narys lentele
CREATE TABLE Narys (
    NarioID INT PRIMARY KEY,
    VartotojoID INT,
    KomandosID INT,
    Pareigos VARCHAR(50),
    
    FOREIGN KEY (VartotojoID) REFERENCES Vartotojas (VartotojoID),
    FOREIGN KEY (KomandosID) REFERENCES Komanda (KomandosID)
);

-- Sukurti Klientas lentele
CREATE TABLE Klientas (
    KlientoID INT PRIMARY KEY,
    Imones_pavadinimas VARCHAR(50),
    Telefono_numeris VARCHAR(15),
    El_pastas VARCHAR(100)
);

-- Sukurti Projektas lentele
CREATE TABLE Projektas (
    ProjektoID INT PRIMARY KEY,
    KomandosID INT,
    KlientoID INT,
    Pavadinimas VARCHAR(50),
    Tikslas VARCHAR(255),
    Aprasymas VARCHAR(255),
    Pradzios_data DATE,
    Pabaigos_data DATE,
    
    FOREIGN KEY (KomandosID) REFERENCES Komanda (KomandosID),
    FOREIGN KEY (KlientoID) REFERENCES Klientas (KlientoID)
);

-- Sukurti Renginys lentele
CREATE TABLE Renginys (
    RenginioID INT PRIMARY KEY,
    OrganizacijosID INT,
    Pavadinimas VARCHAR(50),
    Renginio_data DATE,
    Vieta VARCHAR(50),
    
    FOREIGN KEY (OrganizacijosID) REFERENCES Organizacija (OrganizacijosID)
);

-- Sukurti Resursas lentele
CREATE TABLE Resursas (
    ResursoID INT PRIMARY KEY,
    ProjektoID INT,
    Pavadinimas VARCHAR(50),
    Tipas VARCHAR(50),
    
    FOREIGN KEY (ProjektoID) REFERENCES Projektas (ProjektoID)
);

-- Sukurti Uzduotis lentele
CREATE TABLE Uzduotis (
    UzduotiesID INT PRIMARY KEY AUTO_INCREMENT,
    ProjektoID INT,
    Pavadinimas VARCHAR(50),
    Aprasymas VARCHAR(255),
    Pradzios_data DATE,
    Pabaigos_data DATE,
    Statusas VARCHAR(50),
    NarioID INT,
    
    FOREIGN KEY (ProjektoID) REFERENCES Projektas (ProjektoID),
    FOREIGN KEY (NarioID) REFERENCES Narys (NarioID)
);

-- Sukurti Ataskaita lentele
CREATE TABLE Ataskaita (
    AtaskaitosID INT PRIMARY KEY,
    Pavadinimas VARCHAR(50),
    Sukurimo_data DATE,
    Aprasymas VARCHAR(1000),
    Rezultatas VARCHAR(1000)
);

-- Sukurti Progresas lentele
CREATE TABLE Progresas (
    ProgresasID INT PRIMARY KEY,
    UzduotiesID INT UNIQUE, -- Vienas su vienu rysys su Ataskaita
    AtaskaitosID INT, 
    Stadija VARCHAR(50),
    Procentai VARCHAR(50),
    
    FOREIGN KEY (UzduotiesID) REFERENCES Uzduotis (UzduotiesID),
    FOREIGN KEY (AtaskaitosID) REFERENCES Ataskaita (AtaskaitosID)
);

-- Sukurti Komentaras lentele
CREATE TABLE Komentaras (
    KomentaroID INT PRIMARY KEY,
    UzduotiesID INT,
    VartotojoID INT,
    Pavadinimas VARCHAR(50),
    Tekstas VARCHAR(1000),
    Sukurimo_data DATE,

    FOREIGN KEY (UzduotiesID) REFERENCES Uzduotis (UzduotiesID),
    FOREIGN KEY (VartotojoID) REFERENCES Vartotojas (VartotojoID)
);

-- TRIGGERS

-- Padaro, jog neimanoma istrinti vykdomo projekto
DELIMITER \\
CREATE TRIGGER PreventActiveProjectDeletion
BEFORE DELETE ON Projektas
FOR EACH ROW
BEGIN
    IF EXISTS (
        SELECT 1
        FROM Uzduotis
        WHERE ProjektoID = OLD.ProjektoID AND Statusas = 'Vykdoma'
    ) THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Cannot delete a project with active tasks.';
    END IF;
END;
\\
DELIMITER ;

-- Automatiskai pakeicia uzduoties statusa atlikta, kai ateina pabaigos data
DELIMITER \\
CREATE TRIGGER UpdateTaskStatusOnDate
BEFORE UPDATE ON Uzduotis
FOR EACH ROW
BEGIN
    IF NEW.Pabaigos_data <= CURRENT_DATE() AND NEW.Statusas != 'Atlikta' THEN
        SET NEW.Statusas = 'Atlikta';
    END IF;
END;
\\
DELIMITER ;

-- Automatinis uzduoties statuso atnaujinimas
DELIMITER \\
CREATE TRIGGER UpdateTaskStatusOnAssignment
AFTER UPDATE ON Uzduotis
FOR EACH ROW
BEGIN
    IF NEW.NarioID IS NOT NULL AND OLD.NarioID IS NULL AND NEW.Statusas = 'Nepradeta' THEN
        UPDATE Uzduotis
        SET Statusas = 'Vykdoma'
        WHERE UzduotiesID = NEW.UzduotiesID;
    END IF;
END;
\\
DELIMITER ;

-- Automatinis komentaro sukurimo datos irasymas (GOOD)
DELIMITER \\
CREATE TRIGGER SetDefaultCommentDate
BEFORE INSERT ON Komentaras
FOR EACH ROW
BEGIN
    IF NEW.Sukurimo_data IS NULL THEN
        SET NEW.Sukurimo_data = CURRENT_DATE();
    END IF;
END;
\\
DELIMITER ;
